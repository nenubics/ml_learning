# 🏷️ Навигатор по Тегам и Таксономии

В базе знаний используется строгая трехуровневая иерархическая система тегов в стиле современных Obsidian-хранилищ. Это позволяет мгновенно фильтровать заметки через поиск Obsidian (`tag:#ml/algorithms`), строить графы и генерировать динамические таблицы с помощью плагина **Dataview**.

---

## 🌲 Дерево тегов

```text
#ml
├── /math                     # Математические основы (линал, матан, теорвер)
├── /optimization             # Оптимизационные методы (градиентный спуск, learning rate)
├── /algorithms               # Алгоритмы ML
│   ├── /classical            # Классические методы
│   ├── /regression           # Регрессионные модели
│   ├── /trees                # Деревья решений
│   ├── /ensembles            # Ансамблевые методы (бэггинг, бустинг)
│   └── /boosting             # Градиентный бустинг (XGBoost, GBDT)
├── /cheatsheet               # Шпаргалки и каталоги выбора моделей
├── /data                     # Работа с данными
│   ├── /preprocessing        # Очистка, заполнение пропусков, кодирование
│   └── /feature-engineering  # Создание признаков
├── /data-engineering         # Инженерия данных (БД, Spark, Kafka, Hadoop)
├── /frameworks               # Библиотеки и инструменты
│   ├── /sklearn              # Scikit-learn модели и пайплайны
│   ├── /pandas               # Анализ данных в Pandas
│   ├── /numpy                # Матричные вычисления NumPy
│   ├── /matplotlib           # Визуализация данных
│   └── /pytorch              # Экосистема PyTorch
├── /deep-learning            # Глубокое обучение и нейронные сети
│   └── /tensorflow           # Моделирование на TensorFlow и Keras
├── /resources                # Учебные материалы и ссылки
│   ├── /bootcamp             # Комплексные практические буткемпы
│   ├── /courses              # Курсы и программы
│   ├── /university           # Университетские академические программы
│   ├── /community            # Сообщества и open-source организации
│   ├── /knowledge-base       # Внешние базы знаний
│   └── /roadmap              # Дорожные карты обучения
└── /overview                 # Обзорные материалы и сравнения

#type                         # Тип заметки
├── /concept                  # Теоретический концепт или модель
├── /tool                     # Библиотека или инструмент
├── /tutorial                 # Практический туториал
├── /summary                  # Сводка или сравнительный конспект
├── /roadmap                  # Дорожная карта
└── /resource                 # Внешний источник

#difficulty                   # Сложность
├── /beginner                 # 🟢 Начальный уровень
├── /intermediate             # 🟡 Средний уровень
└── /advanced                 # 🔴 Продвинутый уровень

#status                       # Жизненный цикл заметки
├── /seed                     # 🌰 Заготовка / семя
├── /in-progress              # 🌿 В процессе дополнения
└── /evergreen                # 🌲 Завершенная структурированная заметка
```
